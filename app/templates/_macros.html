{#- Shared template macros. Usage (any template): {% from '_macros.html' import stars, external, evidence_badge %} CSS these macros depend on (keep in sync): .rating .star .rating-count .ext-link -> defined in static/css/playground.css AND static/css/style.css .badge .verified|.reported|.unavailable|.unretrieved -> static/css/playground.css Any page that uses a macro must load one of those stylesheets, or the output renders unstyled. -#} {# Star rating with an accessible label. Anything that is not a number between 0 and 5 is treated as "no rating". #} {% macro stars(rating, count) %} {% if rating is number and 0 < rating <= 5 %} {{ '%.1f'|format(rating) }} {% if count %}({{ count|number }}){% endif %} {% else %} No ratings {% endif %} {% endmacro %} {# External link. Only real http(s) URLs become links; anything else (javascript:, data:, empty) is shown as plain text. #} {% macro external(url, text, css_class='') %} {% if url and (url.startswith('https://') or url.startswith('http://')) %} {{ text }} {% else %} {{ text }} {% endif %} {% endmacro %} {# Evidence status badge: the same four statuses the playground uses everywhere. #} {% macro evidence_badge(status) %} {% set labels = { 'verified': 'Verified at source', 'reported': 'Reported by search · not confirmed', 'unavailable': 'Unavailable', 'unretrieved': 'Could not retrieve' } %} {% if status in labels %}{{ labels[status] }}{% endif %} {% endmacro %}
